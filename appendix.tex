\subsection{Error States}

Below, we formally define error states.

\begin{definition}
\label{def:error-state}
Let $c \triangleq \progstate{\location}{\localstore}{\stack}{\memory}$ be a concrete state.
Then $c$ is an error state, denoted by $\errorstate(c)$, if one of the following holds:
\begin{enumerate}
    \item $\instrat{\location} = \textup{$\codequote{\tunreachable}$}$
    \item $\instrat{\location} = \textup{$\codequote{x \ \texttt{=} \ op \ \bvsort{w} \ e_1 \ e_2}$}$, and one of the following holds (where $v_k \triangleq \evalexprty{e_k}{\localstore}{\bvsort{w}}$):
    \begin{enumerate}[left=-3pt]
        \item \textup{$op \in \{\opcode{udiv}, \opcode{sdiv}, \opcode{urem}, \opcode{srem}\}$ and $v_2 = \bvval{0}{w}$}
        \item \textup{$op = \opcode{sdiv}$, $v_1 = \bvval{(-2^{w - 1})}{w}$, and $v_2 = \bvval{(-1)}{w}$}
        \item \textup{$op \in \{\opcode{shl}, \opcode{lshr}, \opcode{ashr}\}$ and $v_2 = \bvval{n}{w}$, and $n \geq w$}
    \end{enumerate}
    \item $\instrat{\location} = \textup{$\codequote{x \ \texttt{=} \ \tcall^{\pointertypeof{\bvsort{8}}} \ \intrinsicfunc{malloc}(\bvsort{64} \ e)}$}$,
          and \textup{$\evalexprty{e}{\localstore}{\bvsort{64}} > \bvval{(2^{\indexwidth - 1} - 1)}{64}$}.
    \item $\instrat{\location} = \textup{$\codequote{\tcall~\intrinsicfunc{free}(\pointertypeof{\bvsort{8}} \ e)}$}$, \textup{$\evalexprty{e}{\localstore}{\pointertypeof{\bvsort{8}}} = p$},
          and there is no $\mo \in \memory$ such that $\isbaseaddr{p}{\mo}$.
    \item $\instrat{\location} = \textup{$\codequote{x \ \texttt{=} \ \tload^{\tau} \ e}$}$, \textup{$\evalexprty{e}{\localstore}{\pointertypeof{\tau}} = p$},
          and there is no $\mo \in \memory$ such that $\isinbounds{p}{|\tau|}{\mo}$.
    \item $\instrat{\location} = \textup{$\codequote{\tstore^{\tau} \ e_1 \ e_2}$}$, \textup{$\evalexprty{e_2}{\localstore}{\pointertypeof{\tau}} = p$},
          and there is no $\mo \in \memory$ such that $\isinbounds{p}{|\tau|}{\mo}$.
    % TODO: $|\overline{(\tau_i, e_i)}| > 0$
    \item $\instrat{\location} = \textup{$\codequote{x \ \texttt{=} \ \tgep~\pointertypeof{\tau} \ e \ \overline{(\tau_i, e_i)}}$}$ and $\evalgep(\tau, p, \overline{v_i}) = \tpoison$
          (where \textup{$p \triangleq \evalexprty{e}{\localstore}{\pointertypeof{\tau}}$} and $v_i \triangleq \evalexprty{e_i}{\localstore}{\tau_i}$).
\end{enumerate}
\end{definition}

\section{Symbolic Semantics}

\subsection{Symbolic Error States}

In the following, we define when a symbolic state is considered to be an error state.
To do so, we use the following definitions:
\[
\begin{array}{ll}
    \syminvalidpointer(\symmemory, \pc, p, n) & \triangleq \exists m. \ \forall \symmo \in \symmemory. \ m \not \models (\pc \land \rangecond(p, n, \symmo)) \\
    \syminvalidbaseaddress(\symmemory, \pc, p) & \triangleq \exists m. \ \forall \symmo \in \symmemory. \ m \not \models (\pc \land \baseaddrcond(p, \symmo))
\end{array}
\]
Intuitively,
$\syminvalidpointer$ states that there is an SMT model $m$ satisfying $\varphi$ under which $p$ does not belong to any address interval,
and $\syminvalidbaseaddress$ states that under such a model, $p$ does not point to any base address.

\begin{definition}
\label{def:sym-error-state}
A symbolic state $s \triangleq \symstate{\location}{\symlocalstore}{\symstack}{\symmemory}{\pc}$ is an error state, denoted by $\errorsymstate(s)$,
if one of the following holds:
\begin{enumerate}[left=0pt]
    \item $\instrat{\location} = \textup{$\codequote{\tunreachable}$}$
    \item $\instrat{\location} = \textup{$\codequote{x \ \texttt{=} \ op \ \bvsort{w} \ e_1 \ e_2}$}$ and one of the following holds (where $t_j \triangleq \evalexprty{e_j}{\symlocalstore}{\bvsort{w}}$):
    \begin{enumerate}[left=-3pt]
        \item \textup{$op \in \{\opcode{udiv}, \opcode{sdiv}, \opcode{urem}, \opcode{srem}\}$} and $\sat{\pc \land (t_2 = \bvconst{w}{0})}$
        \item \textup{$op = \opcode{sdiv}$} and $\sat{\pc \land (t_1 = \bvconst{w}{(-2^{w - 1})}) \land (t_2 = \bvconst{w}{(-1)})}$
        \item \textup{$op \in \{\opcode{shl}, \opcode{lshr}, \opcode{ashr}\}$} and $\sat{\pc \land (t_2 \geq w)}$
    \end{enumerate}
    \item $\instrat{\location} = \textup{$\codequote{x \ \texttt{=} \ \tcall^{\pointertypeof{\bvsort{8}}} \ \intrinsicfunc{malloc}(\bvsort{64} \ e)}$}$ and
        \textup{$\sat{\pc \land (t > \bvconst{64}{(2^{\indexwidth - 1} - 1)}}$} (where $t \triangleq \evalexprty{e}{\symlocalstore}{\tau}$).
    \item $\instrat{\location} = \textup{$\codequote{\tcall~\intrinsicfunc{free}(\pointertypeof{\bvsort{8}} e)}$}$, \textup{$\evalexprty{e}{\symlocalstore}{\pointertypeof{\bvsort{8}}} = p$}, and $\syminvalidbaseaddress(\symmemory, \pc, p)$.
    \item $\instrat{\location} = \textup{$\codequote{x \ \texttt{=} \ \tload^{\tau} \ e}$}$, \textup{$\evalexprty{e}{\symlocalstore}{\pointertypeof{\tau}} = p$}, and $\syminvalidpointer(\symmemory, \pc, p, |\tau|)$.
    \item $\instrat{\location} = \textup{$\codequote{\tstore^{\tau} \ e_1 \ e_2}$}$, \textup{$\evalexprty{e_2}{\symlocalstore}{\pointertypeof{\tau}} = p$}, and $\syminvalidpointer(\symmemory, \pc, p, |\tau|)$.
    %\item $\instrat{\location} = \textup{$\codequote{x \ \texttt{=} \ \tgep~\pointertypeof{\tau} \ e \ \overline{(\tau_i, e_i)}}$}$ and $\sat{\varphi \land \overflowcond(\tau, \overline{k_i})}$
    %      (where $k_i \triangleq \evalexprty{e_i}{\localstore}{\tau_i}$).
    \item $\instrat{\location} = \textup{$\codequote{x \ \texttt{=} \ \tgep~\pointertypeof{\tau} \ e \ \overline{(\tau_i, e_i)}}$}$ and there exists an SMT model $m$ such that:
    \begin{enumerate}
        \item $m \models \varphi$ and $\evalgep(\tau, p, \overline{v_i}) = \tpoison$ (where $p \triangleq m[\evalexprty{e}{\localstore}{\pointertypeof{\tau}}]$ and $v_i \triangleq m[\evalexprty{e_i}{\localstore}{\tau_i}]$).
    \end{enumerate}
\end{enumerate}
%where $\overflowcond(\tau, \overline{k_i})$ constructs a condition that encodes the fact that there are no overlfows when evaluating the $\tgep$ instruction,
\end{definition}

\subsection{Proof Sketch for \Cref{theorem:safety-via-et}}
\label{appendix:proof-sketch}

To see intuitively why this theorem holds, consider the single-step case.
Let $s$ be a symbolic state and $c$ be a concrete state such that $s \overapprox c$,
and suppose that the subtree $\tree(s, l)$ is safe.
If $c \nsstep c'$, then from~\Cref{theorem:completeness},
it follows that there exists $s'$ such that $s \symstep s'$ and $s' \overapprox c'$.
From~\Cref{def:safe-et}, it follows that $s'$ is not an error state,
and since $s' \overapprox c'$, then $c'$ is not an error state.
Now, let $c_0 \triangleq \initstate(\module)$ be the initial concrete state.
If $\issafens(\module)$, then no error states can be derived from $c_0$.
Next, if $c_0 \step c$,
then based on~\Cref{lemma:multi-ns-step-relative-completeness}, it follows that $c_0 \nsstep c$.
Finally, we know that $c$ is not an error state, so $\issafe(\module)$.
